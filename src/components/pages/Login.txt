export const prerender = false; // WAJIB kalau output: "hybrid" di astro.config

import { comparePassword } from "../../unsued/encrypt/enkripsi";

let errorMessage = "";

if (Astro.request.method === "POST") {
  try {
    const data = await Astro.request.formData();
    const username = data.get("email")?.toString();
    const password = data.get("password")?.toString();

    const userNameEnv = "nakamuraalsut"
    const hashedPasswordEnv = "$2b$12$wOVqo.44hg8okWJoV8UWe.wl.1GDmpQSM9YZnP7shrn10F0eYNxHK"
    const tokenEnv = "verylongpasswordfortoken"

    console.log(userNameEnv)
    console.log(hashedPasswordEnv)
    console.log(tokenEnv)

    // 1. Validasi Environment Variables
    if (!userNameEnv || !hashedPasswordEnv || !tokenEnv) {
      throw new Error(
        "Konfigurasi environment variable (.env) belum lengkap di server.",
      );
    }

    // 2. Validasi Input Form
    if (!username || !password) {
      errorMessage = "Username dan password wajib diisi.";
    } else {
      const isMatch = await comparePassword(password, hashedPasswordEnv);

      if (username !== userNameEnv || !isMatch) {
        errorMessage = "Username atau password salah.";
      } else {
        // Set cookie auth
        Astro.cookies.set("auth_login", tokenEnv, {
          maxAge: 60 * 60,
          path: "/",
          httpOnly: true,
          secure: import.meta.env.PROD,
          sameSite: "lax",
        });

        // PENTING: form submit biasa butuh redirect, bukan JSON response
        return Astro.redirect("/admin/edit", 302);
      }
    }
  } catch (error) {
    // Log error asli ke terminal server supaya bisa didebug
    console.error("Login error:", error);
    errorMessage = "Terjadi kesalahan internal pada server.";
  }
}
